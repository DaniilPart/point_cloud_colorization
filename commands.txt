# pointcloud_colorizer useful commands (split pipeline)

# 1) Build from workspace root
cd ~/ros2_ws
colcon build --packages-select pointcloud_colorizer --symlink-install

# 2) Source workspace
source ~/ros2_ws/install/setup.bash

# 3) Verify package executables
ros2 pkg list | grep pointcloud_colorizer

# 4) Launch the default split pipeline (+ RViz by default)
ros2 launch pointcloud_colorizer colorizers.launch.py

# 5) Launch without RViz
ros2 launch pointcloud_colorizer colorizers.launch.py rviz:=false

# 6) Check output topics
ros2 topic info /colorizer/raw/colored_cloud
ros2 topic hz /colorizer/raw/colored_cloud
ros2 topic info /colorizer/raw/naive_map
ros2 topic hz /colorizer/raw/naive_map

# 7) Helpful runtime checks
ros2 node list
ros2 node info /raw_cloud_colorizer
ros2 node info /colored_cloud_map_aggregator
